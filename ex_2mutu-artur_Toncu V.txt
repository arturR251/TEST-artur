#include <stdio.h>
#include <stdlib.h> // Necesara pentru functia abs()

int main() {
    int m, n;
    // 1. Citirea datelor
    printf("Introduceti primul numar intreg (m): ");
    // Folosim %d pentru citirea unui intreg
    if (scanf("%d", &m) != 1) {
        printf("Eroare la citirea lui m.\n");
        return 1;
    }
    printf("Introduceti al doilea numar intreg (n): ");
    if (scanf("%d", &n) != 1) {
        printf("Eroare la citirea lui n.\n");
        return 1;
    }
    // 2. Verificarea conditiei
    // Doua numere sunt consecutive daca diferenta lor absoluta este 1.
    // Functia abs() returneaza valoarea absoluta a unui intreg.
    if (abs(m - n) == 1) {
        // 3. Afisarea rezultatului (consecutive)
        printf("\nNumerele %d si %d SUNT consecutive.\n", m, n);
    } else {
        // 3. Afisarea rezultatului (NU sunt consecutive)
        printf("\nNumerele %d si %d NU SUNT consecutive.\n", m, n);
    }
    return 0;
}

#include <stdio.h>
#include <math.h>

int main() {
    int n, i;
    printf("Introduceti numarul n: ");
    if (scanf("%d", &n) != 1) return 1;

    if (n <= 1) {
        printf("%d nu este prim.\n", n);
        return 0;
    }
    if (n==2 || n==3) {
        printf("%d este prim.\n", n);
        return 0;
    }
    if (n%2==0 || n%3==0) {
        printf("%d nu este prim.\n", n);
        return 0;
    }
    // Verificarea divizorilor de forma 6k +/- 1 (optim)
    for (i=5; i*i <= n; i=i+6) {
        if (n%i==0 || n%(i+2) == 0) {
            printf("%d nu este prim.\n", n);
            return 0;
        }
    }
    printf("%d este prim.\n", n);
    return 0;
}
